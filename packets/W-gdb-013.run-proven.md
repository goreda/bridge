// AUTHORED NEW (2026-10-25, L3 formats/hardware-decode lane).
// swresample-backed conversion of any L3-admitted decoded audio frame to the
// audio sink contract: packed s16le stereo 44.1 kHz. M1 admission is AAC
// only on this path; broader codec support is not implied by conversion. Conversion context
// is cached per (format, rate, layout) and re-initialized on change.
// Sync policy (when the sink consumes these chunks) stays with L1; this lane
// owns only the format/rate/channel landing.
import Foundation
import CFFmpeg

public final class L3SampleConverter {
    public static let outputRate: Int32 = 44100
    public static let outputChannels = 2

    private var swr: UnsafeMutablePointer<GorAVL3Swr>?
    private(set) public var convertedFrames = 0
    private(set) public var conversionFailures = 0
    /// Duration-conservation witness (blocker 4): input vs output sample
    /// counts let the run report check output duration against the rate
    /// ratio instead of losing resampler delay silently.
    private(set) public var inputSamples: Int64 = 0
    private(set) public var outputSamples: Int64 = 0
    /// v4 (architect): input samples per SOURCE rate - a cross-rate sum
    /// without time normalization is not a duration witness. Duration checks
    /// compare output (fixed 44.1 kHz) against sum_r(input_r * 44100 / r).
    private(set) public var inputSamplesByRate: [Int: Int64] = [:]
    /// Actual re-init count (first init excluded); a reinit drains the old
    /// configuration's delay tail into the next output, never drops it.
    public var reinitCount: Int { Int(gorav_l3_swr_reinits(swr)) }
    /// v4.4 (r6 g3): the resampler's current input-side delay in input
    /// samples at the given input rate (negative = unavailable). The audio
    /// decoder subtracts it when computing an emitted chunk's first-sample
    /// time; the pointer itself stays private.
    public func currentInputDelay(inputRate: Int64) -> Int64 {
        guard let s = swr else { return -1 }
        // A newly allocated converter has no context/buffer until the first
        // convert. Its delay is exactly zero, not an unavailable old phase.
        if s.pointee.ctx == nil { return 0 }
        return gorav_l3_swr_delay(s, inputRate)
    }
    /// v4.2 (blocker 5): seeks the decoder runs through this lane reset the
    /// converter - typed counter for the run report, matching the video
    /// decoder's seekCount.
    private(set) public var seekCount = 0

    public init() {
        swr = gorav_l3_swr_open()
    }
    deinit {
        gorav_l3_swr_close(&swr)
    }

    /// Converts one decoded frame to packed s16le stereo at 44.1 kHz.
    /// Returns (bytes, samplesPerChannel). A zero-sample result (legal for a
    /// drained resampler edge) returns empty bytes with samples == 0 and is
    /// NOT a failure. Any real failure returns nil and is counted.
    public func convert(frame: UnsafeMutablePointer<AVFrame>,
                        maxFrameBytes: Int) -> (bytes: [UInt8], samples: Int)? {
        var outData: UnsafeMutablePointer<UInt8>?
        var outSamples: Int32 = 0
        let rc = gorav_l3_swr_convert(swr, frame, Self.outputRate, &outData, &outSamples)
        guard rc == 0 else {
            conversionFailures += 1
            return nil
        }
        guard outSamples > 0, let data = outData else {
            return ([], 0)
        }
        defer { gorav_l3_av_freep(&outData) }
        let byteCount = Int(outSamples) * Self.outputChannels * 2
        guard byteCount > 0, byteCount <= maxFrameBytes else {
            conversionFailures += 1
            return nil
        }
        convertedFrames += 1
        let inS = Int64(frame.pointee.nb_samples)
        inputSamples += inS
        inputSamplesByRate[Int(frame.pointee.sample_rate), default: 0] += inS
        outputSamples += Int64(outSamples)
        return (Array(UnsafeBufferPointer(start: data, count: byteCount)), Int(outSamples))
    }

    /// v4.2 (blocker 5): drop all resampler state for a seek. The shim
    /// re-initializes the context (delay tail gone, per-format state clean);
    /// the NEXT convert() re-establishes format/rate from the post-seek
    /// frame exactly like the first one. Counted, so the report can pair
    /// converter resets with decoder seeks.
    /// Returns false when the provider reset itself failed - the caller
    /// turns that into a typed seek failure instead of reporting success
    /// over a dirty converter (architect r5).
    @discardableResult
    public func resetForSeek() -> Bool {
        guard swr != nil else { return false }
        gorav_l3_swr_close(&swr)
        swr = gorav_l3_swr_open()
        guard swr != nil else { return false }
        seekCount += 1
        return true
    }

    /// EOF drain (blocker 4): emits the resampler delay tail after the last
    /// decoded frame. Call once at end-of-stream before .endOfStream.
    /// nil = failure; ([], 0) = nothing buffered.
    public func drain(maxFrameBytes: Int) -> (bytes: [UInt8], samples: Int)? {
        var outData: UnsafeMutablePointer<UInt8>?
        var outSamples: Int32 = 0
        let rc = gorav_l3_swr_drain(swr, &outData, &outSamples)
        guard rc == 0 else { conversionFailures += 1; return nil }
        guard outSamples > 0, let data = outData else { return ([], 0) }
        defer { gorav_l3_av_freep(&outData) }
        let byteCount = Int(outSamples) * Self.outputChannels * 2
        guard byteCount > 0, byteCount <= maxFrameBytes else { conversionFailures += 1; return nil }
        outputSamples += Int64(outSamples)
        return (Array(UnsafeBufferPointer(start: data, count: byteCount)), Int(outSamples))
    }
}

import Foundation
import CFFmpeg
import GorAVFFmpegAdapter

public enum ProviderLane {
    public static func layoutMask(_ name: String) -> UInt64 {
        // Layout masks per the FFmpeg channel-layout API used by the product's
        // CFFmpeg module. Newer FFmpeg exposes AVChannelLayout; populate
        // channel_layout/u.mask equivalently. These numeric masks are the
        // long-stable AV_CH_LAYOUT_* values (ASSUMED stable across the
        // FFmpeg major the product pins; verify against the product's
        // CFFmpeg if it differs).
        switch name {
        case "mono": return 0x4
        case "stereo": return 0x3
        case "2.1": return 0x103
        case "3.0": return 0x7
        case "4.0": return 0x33
        case "5.0": return 0x607
        case "5.1": return 0x60f
        case "6.1": return 0x70f
        case "7.1": return 0x63f
        default: return 0 // "unknown" and anything else: no layout mask
        }
    }

    public static func sampleFormat(_ name: String) -> AVSampleFormat? {
        switch name {
        case "u8": return AV_SAMPLE_FMT_U8
        case "s16le": return AV_SAMPLE_FMT_S16
        case "s32le": return AV_SAMPLE_FMT_S32
        case "f32le": return AV_SAMPLE_FMT_FLT
        case "f64le": return AV_SAMPLE_FMT_DBL
        case "u8p": return AV_SAMPLE_FMT_U8P
        case "s16p": return AV_SAMPLE_FMT_S16P
        case "s32p": return AV_SAMPLE_FMT_S32P
        case "fltp": return AV_SAMPLE_FMT_FLTP
        case "dblp": return AV_SAMPLE_FMT_DBLP
        default: return nil
        }
    }

    /// Builds one decoded-frame stand-in: `inSamples` of a sine at `tone`
    /// Hz (or deterministic noise when tone is nil) in the given
    /// layout/format/rate. Layout mask 0 or rate <= 0 leaves the frame
    /// unallocated on purpose: the matrix expects a typed nil-failure there.
    public static func makeFrame(_ d: RefFrame) -> UnsafeMutablePointer<AVFrame>? {
        var f0: UnsafeMutablePointer<AVFrame>? = av_frame_alloc()
        guard let f = f0 else { return nil }
        f.pointee.nb_samples = Int32(d.inSamples)
        // Integrator adaptation (product pins FFmpeg 7.1 headers): the kit was
        // authored against the 4.4-era AVFrame.channel_layout field; the pinned
        // 7.1 API carries AVChannelLayout (ch_layout). Mask values are the same
        // long-stable AV_CH_LAYOUT_* constants.
        av_channel_layout_from_mask(&f.pointee.ch_layout, layoutMask(d.layout))
        f.pointee.sample_rate = Int32(d.inRate)
        guard let fmt = sampleFormat(d.inFormat) else { av_frame_free(&f0); return nil }
        f.pointee.format = fmt.rawValue
        guard f.pointee.ch_layout.nb_channels > 0, d.inRate > 0 else { return f } // unallocated on purpose
        guard av_frame_get_buffer(f, 0) == 0 else { av_frame_free(&f0); return nil }
        let ch = Int(f.pointee.ch_layout.nb_channels)
        let planar = av_sample_fmt_is_planar(fmt) != 0
        for i in 0..<d.inSamples {
            let v = d.tone != nil
                ? sin(2.0 * Double.pi * d.tone! * Double(i) / Double(d.inRate))
                : Double((i * 37) & 0xff) / 128.0 - 1.0
            let planes = planar ? ch : 1
            for p in 0..<planes {
                guard let data = f.pointee.extended_data[p] else { continue }
                let idx = planar ? i : i * ch + p
                switch fmt {
                case AV_SAMPLE_FMT_U8, AV_SAMPLE_FMT_U8P:
                    data[idx] = UInt8(clamping: Int(v * 100.0) + 128)
                case AV_SAMPLE_FMT_S16, AV_SAMPLE_FMT_S16P:
                    data.withMemoryRebound(to: Int16.self, capacity: idx + 1) { $0[idx] = Int16(clamping: Int(v * 12000.0)) }
                case AV_SAMPLE_FMT_S32, AV_SAMPLE_FMT_S32P:
                    data.withMemoryRebound(to: Int32.self, capacity: idx + 1) { $0[idx] = Int32(clamping: Int(v * 800_000_000.0)) }
                case AV_SAMPLE_FMT_FLT, AV_SAMPLE_FMT_FLTP:
                    data.withMemoryRebound(to: Float.self, capacity: idx + 1) { $0[idx] = Float(v * 0.5) }
                case AV_SAMPLE_FMT_DBL, AV_SAMPLE_FMT_DBLP:
                    data.withMemoryRebound(to: Double.self, capacity: idx + 1) { $0[idx] = v * 0.5 }
                default: break
                }
            }
        }
        return f
    }

    public static func freeFrame(_ f: UnsafeMutablePointer<AVFrame>?) {
        var v = f
        av_frame_free(&v)
    }
}

/// One fabricated input frame for the reference lane (no AVFrame, no CFFmpeg).
public struct RefFrame {
    public var layout: String
    public var inFormat: String
    public var inRate: Int
    public var inSamples: Int
    public var tone: Double? // Hz; output is a synthesized s16le sine at this frequency
}

/// Portable reference model of the pinned L3SampleConverter's OBSERVABLE
/// contract (kits/l3-matrix reference lane). It mirrors the pinned excerpt
/// Sources/GorAVFFmpegAdapter/L3SampleConvert.swift (sha256 7910337a, product
/// pin 2112af0d, branch ohal/gorav): same counters, same typed nil-failure
/// rule, same legal zero-sample edge, same delay/drain/seek-reset semantics.
/// Admission rule (layout != unknown, rate > 0, known sample format) was
/// MEASURED against real libswresample in the excerpt harness; see matrix.json
/// provenance. It uses no FFmpeg: it exists so the golden matrix replays
/// anywhere Swift runs, without the gorav_l3_swr_* provider.
public final class ReferenceConverter {
    public static let outputRate = 44100
    public static let outputChannels = 2
    /// Modelled resampler input-side delay, in input samples per active config.
    public static let inputDelay = 32

    public private(set) var convertedFrames = 0
    public private(set) var conversionFailures = 0
    public private(set) var inputSamples: Int64 = 0
    public private(set) var outputSamples: Int64 = 0
    public private(set) var inputSamplesByRate: [Int: Int64] = [:]
    public private(set) var seekCount = 0
    public private(set) var reinitCount = 0

    private struct Config: Equatable { var layout: String; var format: String; var rate: Int }
    private var config: Config?     // nil = no context yet (fresh or post-seek-reset)
    private var cumIn: Int64 = 0    // input samples in current config
    private var emitted: Int64 = 0  // output samples emitted in current config
    private var phase = 0.0
    private var valid = true        // false models provider allocation failure

    public init() {}

    public static let knownFormats: Set<String> =
        ["u8", "s16le", "s32le", "f32le", "f64le", "u8p", "s16p", "s32p", "fltp", "dblp"]

    private func admitted(_ f: RefFrame) -> Bool {
        f.layout != "unknown" && f.inRate > 0 && ReferenceConverter.knownFormats.contains(f.inFormat)
    }

    private func totalIdeal(_ cfg: Config, _ cum: Int64) -> Int64 {
        Int64((Double(cum) * Double(ReferenceConverter.outputRate) / Double(cfg.rate)).rounded(.down))
    }

    /// Mirrors the pinned convert(): nil = typed failure (counted); ([],0) =
    /// legal zero-sample edge (not counted); success counts convertedFrames
    /// and accumulates the duration witnesses.
    public func convert(frame f: RefFrame, maxFrameBytes: Int) -> (bytes: [UInt8], samples: Int)? {
        guard valid else { conversionFailures += 1; return nil }
        guard admitted(f) else { conversionFailures += 1; return nil }
        let cfg = Config(layout: f.layout, format: f.inFormat, rate: f.inRate)
        var out = 0
        if let cur = config, cur != cfg {
            // re-init on (format, rate, layout) change: flush the old delay
            // tail into this output, never drop it (excerpt comment).
            out += Int(totalIdeal(cur, cumIn) - emitted)
            reinitCount += 1
            config = cfg; cumIn = 0; emitted = 0
        } else if config == nil {
            config = cfg; cumIn = 0; emitted = 0
        }
        cumIn += Int64(f.inSamples)
        let delayOut = Int64((Double(ReferenceConverter.inputDelay) * Double(ReferenceConverter.outputRate) / Double(cfg.rate)).rounded(.up))
        let avail = max(Int64(0), totalIdeal(cfg, cumIn) - delayOut)
        out += Int(avail - emitted)
        emitted = avail
        let byteCount = out * ReferenceConverter.outputChannels * 2
        guard byteCount > 0 else { return ([], 0) }
        guard byteCount <= maxFrameBytes else { conversionFailures += 1; return nil }
        convertedFrames += 1
        inputSamples += Int64(f.inSamples)
        inputSamplesByRate[f.inRate, default: 0] += Int64(f.inSamples)
        outputSamples += Int64(out)
        return (synth(tone: f.tone, samples: out), out)
    }

    /// Mirrors the pinned drain(): emits the modelled delay tail.
    public func drain(maxFrameBytes: Int) -> (bytes: [UInt8], samples: Int)? {
        guard valid else { conversionFailures += 1; return nil }
        guard let cfg = config else { return ([], 0) }
        let out = Int(totalIdeal(cfg, cumIn) - emitted)
        emitted += Int64(out)
        let byteCount = out * ReferenceConverter.outputChannels * 2
        guard byteCount > 0 else { return ([], 0) }
        guard byteCount <= maxFrameBytes else { conversionFailures += 1; return nil }
        outputSamples += Int64(out)
        return (synth(tone: nil, samples: out), out)
    }

    /// Mirrors the pinned resetForSeek(): drops all resampler state (delay
    /// tail gone), keeps the counters, bumps seekCount, returns success.
    @discardableResult
    public func resetForSeek() -> Bool {
        guard valid else { return false }
        config = nil; cumIn = 0; emitted = 0
        seekCount += 1
        return true
    }

    /// Mirrors the pinned currentInputDelay(): -1 unavailable, 0 before the
    /// first convert establishes a context, otherwise the modelled delay.
    public func currentInputDelay(inputRate: Int64) -> Int64 {
        guard valid else { return -1 }
        guard config != nil else { return 0 }
        return Int64(ReferenceConverter.inputDelay)
    }

    private func synth(tone: Double?, samples: Int) -> [UInt8] {
        let f = tone ?? 440.0
        var bytes = [UInt8](); bytes.reserveCapacity(samples * 4)
        for _ in 0..<samples {
            let v = Int16(clamping: Int(sin(2.0 * Double.pi * f * phase / Double(ReferenceConverter.outputRate)) * 12000.0))
            phase += 1
            bytes.append(UInt8(truncatingIfNeeded: v))
            bytes.append(UInt8(truncatingIfNeeded: v >> 8))
            bytes.append(UInt8(truncatingIfNeeded: v))
            bytes.append(UInt8(truncatingIfNeeded: v >> 8))
        }
        return bytes
    }
}

            case "nil-failure": self = .nilFailure
            case "empty": self = .empty

final class RefConv: Conv {
    let inner = ReferenceConverter()

    var conversionFailures: Int { inner.conversionFailures }

    }
    func drain(maxFrameBytes: Int) -> (bytes: [UInt8], samples: Int)? {

}

func goertzel(_ s16: [Int16], rate: Double, freq: Double) -> Double {
    guard !s16.isEmpty else { return 0 }
    let w = 2.0 * Double.pi * freq / rate
    let coeff = 2.0 * cos(w)
    var s0 = 0.0, s1 = 0.0, s2 = 0.0
    for x in s16 { s0 = Double(x) + coeff * s1 - s2; s2 = s1; s1 = s0 }
    return (s1 * s1 + s2 * s2 - coeff * s1 * s2) / Double(s16.count * s16.count)
}

func leftChannel(_ bytes: [UInt8]) -> [Int16] {
    var out: [Int16] = []
    out.reserveCapacity(bytes.count / 4)
    var i = 0
    while i + 1 < bytes.count {
        out.append(Int16(bitPattern: UInt16(bytes[i]) | (UInt16(bytes[i + 1]) << 8)))
        i += 4 // packed stereo: left sample only
    }
    return out
}

func runConvertRow(_ row: MatrixDoc.Row, lane: String, make: () -> Conv) {
    let c = make()
    let d = RefFrame(layout: row.layout ?? "", inFormat: row.inFormat ?? "",
                     inRate: row.inRate ?? 0, inSamples: row.inSamples ?? 1024, tone: nil)
    let failsBefore = c.conversionFailures
    let r = c.convert(frame: d, maxFrameBytes: 1 << 22)
    switch row.expect {
    case .converted:
        record(r != nil && r!.samples > 0, row.id, lane,
               "expected converted, got \(r == nil ? "nil" : "samples=\(r!.samples)")")
        if r != nil { record(c.conversionFailures == failsBefore, row.id, lane, "converted but conversionFailures incremented") }
    case .nilFailure:
        record(r == nil, row.id, lane, "expected nil-failure, got samples=\(r?.samples ?? -1)")
        record(c.conversionFailures == failsBefore + 1, row.id, lane,
               "nil-return must be counted in conversionFailures (delta=\(c.conversionFailures - failsBefore))")
    default:
        record(false, row.id, lane, "bad convert-row expect")
    }
}

func runSequence(_ row: MatrixDoc.Row, lane: String, make: () -> Conv) {
    let c = make()
    var okAll = true
    var postResetTone: Double? = nil
    var preResetTone: Double? = nil
    var lastBytes: [UInt8] = []
    func check(_ ok: Bool, _ detail: String) { if !ok { okAll = false; print("FAIL \(row.id) [\(lane)] \(detail)") } }
    for step in row.steps ?? [] {
        switch step.op {
        case "convert":
            let d = RefFrame(layout: step.layout ?? "", inFormat: step.inFormat ?? "",
                             inRate: step.inRate ?? 0, inSamples: step.inSamples ?? 1024, tone: step.tone)
            for _ in 0..<(step.`repeat` ?? 1) {
                let failsBefore = c.conversionFailures
                let r = c.convert(frame: d, maxFrameBytes: step.maxFrameBytes ?? (1 << 22))
                switch step.expect {
                case .converted:
                    check(r != nil, "convert \(d.layout)/\(d.inFormat)/\(d.inRate) expected converted, got nil")
                    if let r = r { lastBytes = r.bytes }
                case .nilFailure:
                    check(r == nil, "convert expected nil-failure, got samples=\(r?.samples ?? -1)")
                    check(c.conversionFailures == failsBefore + 1, "nil-return not counted in conversionFailures")
                default: check(false, "convert step missing expect")
                }
            }
        case "drain":
            let r = c.drain(maxFrameBytes: step.maxFrameBytes ?? (1 << 22))
            switch step.expect {
            case .converted: check(r != nil && r!.samples > 0, "drain expected tail, got \(r?.samples ?? -1)")
            case .empty: check(r != nil && r!.samples == 0, "drain expected empty, got \(r?.samples ?? -1)")
            case .nilFailure: check(r == nil, "drain expected nil-failure")
            default: check(false, "drain step missing expect")
            }
        case "resetForSeek":
            preResetTone = postResetTone ?? preResetTone
            let ok = c.resetForSeek()
            check(ok == (step.expect == .bool(true)), "resetForSeek returned \(ok)")
        case "checkDelay":
            let dl = c.currentInputDelay(inputRate: Int64(step.inputRate ?? 44100))
            switch step.rule {
            case "==0": check(dl == 0, "delay \(dl) != 0")
            case ">=0": check(dl >= 0, "delay \(dl) < 0")
            default: check(false, "unknown delay rule")
            }
        case "checkCounters":
            if let v = step.convertedFrames { check(c.convertedFrames == v, "convertedFrames \(c.convertedFrames) != \(v)") }
            if let v = step.conversionFailures { check(c.conversionFailures == v, "conversionFailures \(c.conversionFailures) != \(v)") }
            if let v = step.seekCount { check(c.seekCount == v, "seekCount \(c.seekCount) != \(v)") }
            if let v = step.reinitCount { check(c.reinitCount == v, "reinitCount \(c.reinitCount) != \(v)") }
        case "checkDuration":
            var ideal = 0.0
            for (rate, n) in c.inputSamplesByRate { ideal += Double(n) * 44100.0 / Double(rate) }
            let tol = step.tolerance ?? 0.005
            let delta = abs(Double(c.outputSamples) - ideal)
            check(delta <= max(64.0, ideal * tol), "duration drift: out=\(c.outputSamples) ideal=\(ideal) delta=\(delta)")
        case "checkNoPreResetTone":
            // Witness: the post-reset output carries NO energy at the
            // pre-reset tone frequency relative to the post-reset tone.
            let s16 = leftChannel(lastBytes)
            let pre = step.freq ?? 440
            var best = 0.0, bestP = 0.0
            for f in stride(from: 220.0, through: 1760.0, by: 220.0) where abs(f - pre) > 1 {
                let p = goertzel(s16, rate: 44100, freq: f)
                if p > bestP { bestP = p; best = f }
            }
            let pPre = goertzel(s16, rate: 44100, freq: pre)
            let ratio = bestP > 0 ? pPre / bestP : Double.infinity
            check(bestP > 0 && ratio <= (step.maxRatio ?? 0.001),
                  "pre-reset tone \(pre)Hz not eliminated: power=\(pPre) vs \(best)Hz power=\(bestP) ratio=\(ratio)")
            postResetTone = best
        default:
            check(false, "unknown op \(step.op)")
        }
    }
    record(okAll, row.id, lane, "sequence \(row.name ?? "") failed")
}
