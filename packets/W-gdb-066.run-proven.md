```swift id=ex_effective_video_rect
public func effectiveVideoRect(drawableW: Int, drawableH: Int, frameW: Int, frameH: Int, _ t: HostTransform) -> VideoRect? {
    guard drawableW > 0, drawableH > 0, frameW > 0, frameH > 0 else { return nil }
    let dw = Double(drawableW), dh = Double(drawableH)
    let s = min(dw / Double(frameW), dh / Double(frameH))
    let w = Double(frameW) * s * t.zoom, h = Double(frameH) * s * t.zoom
    let cx = dw / 2 + t.panX * dw
    let cy = dh / 2 - t.panY * dh
    return VideoRect(x: cx - w / 2, y: cy - h / 2, w: w, h: h)
}
```

```swift id=fx_host_window_geometry
// d9 host-geometry gate (PLAN.md P9): the host-window mode machine and its
// geometry asserts in production code, so the e1 gor-seam run's
// attach/resize/fullscreen/PiP legs assert against THESE predicates rather
// than test-side arithmetic. Content placement reuses the pinned
// effectiveVideoRect (keepaspect letterbox, identity transform). Pure Swift,
// no AppKit: verifiable off-Mac. AUTHORED NEW 2026-11-16, M2-picture/window
// lane (arch-196z item 3).

public enum HostWindowMode: String, Equatable, Sendable { case windowed, fullscreen, pip }

public enum HostWindowRefusal: String, Equatable, Sendable {
    case invalidHostSize        // non-positive host dimension
    case invalidFrameSize       // non-positive frame dimension
    case belowMinimumHost       // under the minimum usable host size
    case notWindowed            // enter* requires the windowed mode
    case alreadyWindowed        // exit* requires the matching non-windowed mode
}

/// Host-window geometry state for the d9 gate: mode, host size, the windowed
/// size to restore, and the keepaspect content rect for the current host.
public struct HostWindowGeometry: Equatable, Sendable {
    public static let minHostWidth = 64
    public static let minHostHeight = 64

    public private(set) var mode: HostWindowMode
    public let frameWidth: Int, frameHeight: Int
    public private(set) var hostWidth: Int, hostHeight: Int
    public private(set) var restoreWidth: Int, restoreHeight: Int

    /// Attach at a windowed host size. nil on invalid or below-minimum
    /// geometry: a host the gate cannot reason about refuses, never clamps.
    public static func attach(hostW: Int, hostH: Int, frameW: Int, frameH: Int) -> HostWindowGeometry? {
        guard hostW > 0, hostH > 0 else { return nil }
        guard frameW > 0, frameH > 0 else { return nil }
        guard hostW >= minHostWidth, hostH >= minHostHeight else { return nil }
        return HostWindowGeometry(mode: .windowed, frameWidth: frameW, frameHeight: frameH,
                                  hostWidth: hostW, hostHeight: hostH,
                                  restoreWidth: hostW, restoreHeight: hostH)
    }

    /// Keepaspect content rect in the current host, identity transform.
    /// Non-nil for every state attach/resize admits.
    public var contentRect: VideoRect? {
        effectiveVideoRect(drawableW: hostWidth, drawableH: hostHeight,
                           frameW: frameWidth, frameH: frameHeight, .identity)
    }

    /// Resize the host in any mode. On refusal the state is unchanged.
    @discardableResult
    public mutating func resize(hostW: Int, hostH: Int) -> HostWindowRefusal? {
        guard hostW > 0, hostH > 0 else { return .invalidHostSize }
        guard hostW >= Self.minHostWidth, hostH >= Self.minHostHeight else { return .belowMinimumHost }
        hostWidth = hostW; hostHeight = hostH
        if mode == .windowed { restoreWidth = hostW; restoreHeight = hostH }
        return nil
    }

    /// Enter fullscreen at the screen size; the windowed size is kept for
    /// restore. Requires .windowed.
    @discardableResult
    public mutating func enterFullscreen(hostW: Int, hostH: Int) -> HostWindowRefusal? {
        guard mode == .windowed else { return .notWindowed }
        guard hostW > 0, hostH > 0 else { return .invalidHostSize }
        guard hostW >= Self.minHostWidth, hostH >= Self.minHostHeight else { return .belowMinimumHost }
        mode = .fullscreen; hostWidth = hostW; hostHeight = hostH
        return nil
    }

    /// Exit fullscreen back to the saved windowed size.
    @discardableResult
    public mutating func exitFullscreen() -> HostWindowRefusal? {
        guard mode == .fullscreen else { return .alreadyWindowed }
        mode = .windowed; hostWidth = restoreWidth; hostHeight = restoreHeight
        return nil
    }

    /// Enter PiP at the pip window size; the windowed size is kept for
    /// restore. Requires .windowed.
    @discardableResult
    public mutating func enterPiP(hostW: Int, hostH: Int) -> HostWindowRefusal? {
        guard mode == .windowed else { return .notWindowed }
        guard hostW > 0, hostH > 0 else { return .invalidHostSize }
        guard hostW >= Self.minHostWidth, hostH >= Self.minHostHeight else { return .belowMinimumHost }
        mode = .pip; hostWidth = hostW; hostHeight = hostH
        return nil
    }

    /// Exit PiP back to the saved windowed size.
    @discardableResult
    public mutating func exitPiP() -> HostWindowRefusal? {
        guard mode == .pip else { return .alreadyWindowed }
        mode = .windowed; hostWidth = restoreWidth; hostHeight = restoreHeight
        return nil
    }
}
```

```swift id=fx_host_geometry_tests
func rectNear(_ r: VideoRect?, _ x: Double, _ y: Double, _ w: Double, _ h: Double) -> Bool {
    guard let r = r else { return false }
    return near(r.x, x) && near(r.y, y) && near(r.w, w) && near(r.h, h)

// Attach and keepaspect content placement (16:9 frame).
var g = HostWindowGeometry.attach(hostW: 800, hostH: 600, frameW: 1920, frameH: 1080)!
check("H1 attach windowed, letterbox content", g.mode == .windowed && rectNear(g.contentRect, 0, 75, 800, 450))

// Resize in windowed mode; restore size tracks.

      g.resize(hostW: 32, hostH: 300) == .belowMinimumHost && g.hostWidth == 400 && g.hostHeight == 300 &&
      g.resize(hostW: 0, hostH: 300) == .invalidHostSize && g.hostWidth == 400)

// Fullscreen round trip.

check("H8 exit pip restores the windowed size", g.exitPiP() == nil && g.mode == .windowed && g.hostWidth == 400 && g.hostHeight == 300)

// Typed transition refusals.

check("H10 mismatched exit refused", g.exitFullscreen() == .alreadyWindowed && g.exitPiP() == .alreadyWindowed && g.mode == .windowed)

check("H11 invalid or tiny attach refused", HostWindowGeometry.attach(hostW: 0, hostH: 600, frameW: 1920, frameH: 1080) == nil &&

// The d9 keepaspect assert: content aspect matches the frame across a host sweep.

        if let r = gg.contentRect { if abs((r.w / r.h) - 16.0 / 9.0) > 0.01 { hold = false } } else { hold = false }
    }

check("H12 keepaspect holds across the host sweep in fullscreen", hold)

_ = pg.enterPiP(hostW: 320, hostH: 180)
_ = pg.resize(hostW: 400, hostH: 240)
check("H13 exit PiP restores the windowed size, not the PiP size", pg.exitPiP() == nil && pg.mode == .windowed && pg.hostWidth == 800 && pg.hostHeight == 600)
check("H14 mismatched exits refuse typed and keep state", pg.exitPiP() == .alreadyWindowed && pg.exitFullscreen() == .alreadyWindowed && pg.mode == .windowed && pg.hostWidth == 800)

exit(failures == 0 ? 0 : 1)
```
