## problem (what fails for the user)
G0762 routing: the duplicate-adopt -> refused-offer -> abandon sequence can
free bytes belonging to a live lease; the consuming decoder owner must settle
no-reuse/duplicate/foreign-session/abandon policies and add a discriminating
test before any integration. This packet settles the policies in
RetainedPixelOwner and proves them with the discriminating sequence.

## settled policies
- no-reuse: adopt(handle:bytes:) refuses a handle whose bytes are still live;
  a live lease's bytes can never be clobbered by a duplicate adopt.
- duplicate: a refused adopt transfers nothing, so the caller owns nothing to
  abandon for that handle; duplicateAdoptRefused counts the attempt.
- abandon: frees only caller-owned bytes (an accepted adopt whose offer was
  refused); it never frees bytes the store will release through the hook.
- foreign-session: frames of another session never enter through this owner;
  the store refuses foreign leases typed (.foreignSession).

## discriminating test
retained-owner-policy-tests (G1..G7): adopt+offer a lease, duplicate adopt
(refused, nothing clobbered), duplicate offer (typed .duplicateLease
refusal), abandon attempt on the live handle (frees nothing), pinned access
still reads the original bytes, close releases them exactly once, ledger
balances.

'retained-owner-policy-tests: 0 FAIL'.

```swift id=ex_refusal
public enum FrameLeaseRefusal: Error, Equatable, Sendable {
    /// Frame session does not match the store's session.
    case foreignSession
    /// Store is closed; nothing can be retained, pinned, accessed or released.
    case storeClosed
    /// No live lease known for this handle (never retained, or already
    /// logically released by overflow/fence/close).
    case unknownLease
    /// Caller's or lease's seek generation is behind the store's fence.
    case staleGeneration(current: UInt64)
    /// Release or access of a lease that was already explicitly released.
    case alreadyReleased
    /// A lease for this handle is already outstanding (including one whose
    /// owner hook is still deferred behind pins).
    case duplicateLease
    /// Single frame exceeds the byte budget even when empty, or a negative
    /// byte count was claimed.
    case exceedsCapacity
    /// Overflow could not be resolved: every evictable candidate is pinned.

    /// unpin without a matching outstanding pin.

    /// The caller offered a frame without the generation the lease needs
```

```swift id=ex_tap_offerleased
    @discardableResult
    public mutating func offerLeased(_ frame: NativeFrame, clockNow: Double, seekGeneration: UInt64) -> Verdict {
        guard let store = leaseStore else { return offer(frame, clockNow: clockNow) }
        guard frame.session == session else {
            counters.refusedStaleSession += 1
            return .refusedStaleSession
        }
        if minInterval > 0, clockNow.isFinite, let last = lastAcceptClock, clockNow - last < minInterval {
            counters.droppedRateLimit += 1
            return .droppedRateLimit
        }
        switch store.retain(frame, bytes: byteEstimate?(frame) ?? 0, seekGeneration: seekGeneration) {
        case .failure(let refusal):
            counters.refusedLease += 1
            return .refusedLease(refusal)
        case .success:
            break
        }
        var verdict: Verdict = .accepted
        if queue.count >= capacity {
            let evicted = queue.removeFirst()
            counters.evictedOldest += 1
            verdict = .acceptedEvictingOldest
            // Bounded overflow logically releases the evicted frame's lease
            // (owner hook deferred while that lease is pinned).
            _ = store.release(evicted.handle)
        }
        queue.append(frame)
        newest = frame
        if clockNow.isFinite { lastAcceptClock = clockNow }
        counters.accepted += 1
        return verdict
    }

        if newest != nil { counters.readbacks += 1 }
```

```swift id=fx_owner_patch
import Foundation

/// Retained-byte owner for the frame-lease path (settled lease policies,
/// G0761): the decoder frame producer adopts bytes before offering a lease;
/// the lease store's onRelease hook frees them exactly once. Policies:
///  - no-reuse: adopt refuses a handle whose bytes are still live (a live
///    lease's bytes can never be clobbered by a duplicate adopt);
///  - duplicate: a refused adopt transfers nothing, so the caller holds no
///    owner bytes to abandon for that handle;
///  - abandon frees only bytes the caller actually owns (a previously
///    accepted adopt whose offer was refused);
///  - foreign-session frames never enter through this owner: the session is
///    fixed at init and the store refuses foreign leases typed.
/// Foundation only; calls no GorAV API.
public final class RetainedPixelOwner: @unchecked Sendable {
    private let lock = NSLock()
    public let session: UInt64
    private var live: [UInt64: [UInt8]] = [:]
    private var released: [UInt64] = []
    private var abandoned: [UInt64] = []
    private var unknown = 0
    private var doubles = 0
    private var adopted = 0
    private var duplicateAdopts = 0
    private var leased: Set<UInt64> = []

    public init(session: UInt64) { self.session = session }

    /// No-reuse: true when the bytes moved into the owner; false (nothing
    /// stored, nothing freed) when the handle is already live.
    @discardableResult
    public func adopt(handle: UInt64, bytes: [UInt8]) -> Bool {
        lock.lock(); defer { lock.unlock() }
        guard live[handle] == nil else { duplicateAdopts += 1; return false }
        live[handle] = bytes
        adopted += 1
        return true
    }

    /// Caller protocol: after offerLeased accepts, the producer marks the
    /// handle leased; a leased handle's bytes belong to the store's lease and
    /// only the releaseHook frees them.
    public func noteLeaseAccepted(handle: UInt64) {
        lock.lock(); defer { lock.unlock() }
        if live[handle] != nil { leased.insert(handle) }
    }

    /// Frees bytes the caller owns (adopted, never leased). Refuses a handle
    /// under a live lease: those bytes are freed exactly once by the store's
    /// releaseHook, never by abandon.
    @discardableResult
    public func abandon(handle: UInt64) -> Bool {
        lock.lock(); defer { lock.unlock() }
        guard live[handle] != nil, !leased.contains(handle) else { return false }
        live.removeValue(forKey: handle)
        abandoned.append(handle)
        return true
    }

    /// Exactly the onRelease parameter type of FrameLeaseStore.init. The store
    /// fires it once per retained lease, without its lock held; the owner must
    /// never call back into the store.
    public var releaseHook: (UInt64, UInt64) -> Void {
        return { [self] handle, _ in
            self.lock.lock(); defer { self.lock.unlock() }
            if self.live.removeValue(forKey: handle) != nil {
                self.leased.remove(handle)
                self.released.append(handle)
            } else if self.released.contains(handle) || self.abandoned.contains(handle) {
                self.doubles += 1
            } else {
                self.unknown += 1
            }
        }
    }

    public func firstByte(handle: UInt64) -> UInt8? {
        lock.lock(); defer { lock.unlock() }
        return live[handle]?.first
    }

    public var liveHandles: [UInt64] { lock.lock(); defer { lock.unlock() }; return live.keys.sorted() }
    public var releasedHandles: [UInt64] { lock.lock(); defer { lock.unlock() }; return released.sorted() }
    public var abandonedHandles: [UInt64] { lock.lock(); defer { lock.unlock() }; return abandoned.sorted() }
    public var liveBytes: Int { lock.lock(); defer { lock.unlock() }; return live.values.reduce(0) { $0 + $1.count } }
    public var unknownReleases: Int { lock.lock(); defer { lock.unlock() }; return unknown }
    public var doubleReleases: Int { lock.lock(); defer { lock.unlock() }; return doubles }
    public var adoptedCount: Int { lock.lock(); defer { lock.unlock() }; return adopted }
    public var duplicateAdoptRefused: Int { lock.lock(); defer { lock.unlock() }; return duplicateAdopts }
}
```

```swift id=fx_policy_tests
check("G2 duplicate adopt counted", owner.duplicateAdoptRefused == 1)
check("G2 live bytes still the first adopt's", owner.firstByte(handle: 1) == 0xA1 && owner.liveBytes == 16)

// G3: a second offer of the same handle refuses typed duplicateLease.
let dupOffer = tap.offerLeased(frame(7, 1), clockNow: .nan, seekGeneration: 0)
var dupRefused = false
if case .refusedLease(.duplicateLease) = dupOffer { dupRefused = true }

tap.closeLeases()
```
