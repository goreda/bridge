    public static func luma(_ f: LabCropFrame, x: Int, y: Int) -> Int {
        let p0 = f.planes[0]
        let base = y * f.strides[0]
        switch f.format {
        case .gray8, .yuv420p, .nv12:
            var v = Int(p0[base + x])
            if f.range == .limited {
                v = (v - 16) * 255 / 219
                if v < 0 { v = 0 }; if v > 255 { v = 255 }
            }
            return v
        case .rgb24:
            let q = base + 3 * x
            return (54 * Int(p0[q]) + 183 * Int(p0[q + 1]) + 19 * Int(p0[q + 2]) + 128) >> 8
        case .rgba:
            let q = base + 4 * x
            return (54 * Int(p0[q]) + 183 * Int(p0[q + 1]) + 19 * Int(p0[q + 2]) + 128) >> 8
        case .bgra:
            let q = base + 4 * x
            return (54 * Int(p0[q + 2]) + 183 * Int(p0[q + 1]) + 19 * Int(p0[q]) + 128) >> 8
        }
    }

var failures = 0
func check(_ n: String, _ ok: Bool) { print(ok ? "PASS \(n)" : "FAIL \(n)"); if !ok { failures += 1 } }

func gray(_ w: Int, _ h: Int, _ v: UInt8, range: LabCropRange = .full, stride s: Int? = nil) -> LabCropFrame {
    LabCropFrame(planes: [[UInt8](repeating: v, count: (s ?? w) * h)], strides: [s ?? w], width: w, height: h, format: .gray8, range: range)!
}
func grayBox(_ w: Int, _ h: Int, _ x0: Int, _ x1: Int, _ y0: Int, _ y1: Int, _ v: UInt8 = 200) -> LabCropFrame {
    var p = [UInt8](repeating: 0, count: w * h)
    for y in y0..<y1 { for x in x0..<x1 { p[y * w + x] = v } }
    return LabCropFrame(planes: [p], strides: [w], width: w, height: h, format: .gray8, range: .full)!
}
func packed(_ r: UInt8, _ g: UInt8, _ b: UInt8, format: LabCropPixelFormat) -> LabCropFrame {
    var p: [UInt8]
    switch format {
    case .rgb24: p = [r, g, b]
    case .rgba:  p = [r, g, b, 255]
    case .bgra:  p = [b, g, r, 255]
    default:     p = [r]
    }
    return LabCropFrame(planes: [p], strides: [p.count], width: 1, height: 1, format: format, range: .full)!
}
