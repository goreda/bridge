```swift id=fx_display_rotation
// Display-size rotation + pixel aspect (P28 rotation, no architect spec):
// semantics derived from gor-fit.lua@d8b677b (internal source, not bridged):
// the demux track's pixel aspect stretches width (par > 1) or height
// (par < 1) BEFORE rotation is considered (lines 78-82), and a rotation with
// rotation mod 180 == 90 swaps the display width and height (lines 83-84;
// the same swap drives display-follow at lines 215-227, which deliberately
// does not assume the VO rotates its frames). Pure Swift, no AppKit.
// AUTHORED NEW 2026-11-16, M2-picture/window lane (arch-196z item 3).
import Foundation

/// Effective display size from a demuxed video track: pixel aspect applied
/// first, then the quarter-turn swap. nil on a non-positive demux size.
/// A non-finite or non-positive par is ignored (no stretch), matching the
/// source rule that only par > 0 and par != 1 scales.
public func gorDisplaySize(demuxW: Int, demuxH: Int, par: Double = 1, rotation: Int = 0) -> (w: Int, h: Int)? {
    guard demuxW > 0, demuxH > 0 else { return nil }
    var w = Double(demuxW), h = Double(demuxH)
    if par.isFinite, par > 0, par != 1 {
        if par > 1 { w = w * par } else { h = h / par }
    }
    // Lua % is floored; Swift % truncates. Normalize so negative rotations
    // follow the source rule (rotation mod 180 == 90 swaps).
    let r = ((rotation % 180) + 180) % 180
    if r == 90 { swap(&w, &h) }
    let iw = Int(w.rounded()), ih = Int(h.rounded())
    guard iw > 0, ih > 0 else { return nil }
    return (iw, ih)
}
```

```swift id=fx_display_rotation_tests
// Sources/display-rotation-tests/main.swift  (exit 0 = all pass)
// M2-picture/window lane (arch-196z item 3): rotation + pixel-aspect display
// size harness. Derived from gor-fit.lua@d8b677b, no architect spec.
import GorAVPlaybackCore
#if canImport(Glibc)
import Glibc
#else
import Darwin
#endif

var failures = 0
func check(_ n: String, _ ok: Bool) { print(ok ? "PASS \(n)" : "FAIL \(n)"); if !ok { failures += 1 } }
func dimsEq(_ r: (w: Int, h: Int)?, _ w: Int, _ h: Int) -> Bool { r != nil && r!.w == w && r!.h == h }

// Quarter turns swap; half and full turns do not.
check("R1 rotation 90 swaps", dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: 90), 1080, 1920))
check("R2 rotation 270 swaps (270 mod 180 == 90)", dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: 270), 1080, 1920))
check("R3 rotation 180 does not swap", dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: 180), 1920, 1080))
check("R4 rotation 0 and 360 do not swap", dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: 0), 1920, 1080) && dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: 360), 1920, 1080))
check("R5 negative quarter turn swaps (floored modulo)", dimsEq(gorDisplaySize(demuxW: 1920, demuxH: 1080, rotation: -90), 1080, 1920))

// Pixel aspect applies BEFORE rotation.
check("R6 anamorphic par 2 stretches width", dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: 2), 1440, 576))
check("R7 par below 1 stretches height", dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: 0.5), 720, 1152))
check("R8 par then rotation: swap after the stretch", dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: 2, rotation: 90), 576, 1440))
check("R9 par 1 and non-positive par leave the size alone", dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: 1), 720, 576) && dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: 0), 720, 576) && dimsEq(gorDisplaySize(demuxW: 720, demuxH: 576, par: .nan), 720, 576))

// Typed refusals.
check("R10 non-positive demux size refuses", gorDisplaySize(demuxW: 0, demuxH: 1080) == nil && gorDisplaySize(demuxW: 1920, demuxH: -1) == nil)

print(failures == 0 ? "display-rotation-tests: all pass" : "display-rotation-tests: \(failures) FAIL")
exit(failures == 0 ? 0 : 1)
```
