// P28 display-follow rotation handling + keepaspect-window release/restore,
// ported from gor-fit.lua@d8b677b (internal source, not bridged). Semantics
// by source line: the keepaspect-window lock is captured once, released at
// most once for a motion and restored to the captured value exactly once
// when the motion ends or is cancelled (lines 99-107, 136, 163); the
// display-follow decision re-reads video-params, applies the quarter-turn
// swap (lines 226-228), keeps a 2% aspect deadband and never recenters a
// fresh file (lines 229-232), defers to crop ownership, fullscreen, missing
// metrics and the gor-owned window flags (lines 221-223, 233-240), and
// retries after an in-flight animation instead of replaying stale state
// (lines 241-245). Pure Swift, no AppKit: the mpv property writes are
// returned as commands. AUTHORED NEW 2026-11-16, M2-picture/window lane
// (arch-196z item 3).
import Foundation

/// Command the host must issue to mpv for the keepaspect-window lock.
public enum KawCommand: Equatable, Sendable {
    case setNo
    case restoreOriginal
}

/// keepaspect-window release/restore state machine. The captured original is
/// immutable; release/restore are each idempotent, so a settle and a cancel
/// landing together still produce exactly one restore.
public struct KawLock: Equatable, Sendable {
    public let originalKeepsAspect: Bool
    public private(set) var released: Bool
    public init(originalKeepsAspect: Bool) {
        self.originalKeepsAspect = originalKeepsAspect
        self.released = false
    }
    @discardableResult
    public mutating func release() -> KawCommand? {
        if released { return nil }
        released = true
        return .setNo
    }
    @discardableResult
    public mutating func restore() -> KawCommand? {
        if !released { return nil }
        released = false
        return .restoreOriginal
    }
}

/// Everything the display-follow decision reads, snapshot by the host.
public struct FollowInput: Equatable, Sendable {
    public var windowTransitionActive = false
    public var pipWindowOwned = false
    public var stereoWindowOwned = false
    public var demuxW = 0, demuxH = 0, rotate = 0
    public var lastAspect: Double? = nil
    public var fileFresh = false
    public var cropActive = false
    public var gorcropPresent = false
    public var fullscreen = false
    public var metricsValid = false
    /// Seconds still owned by an in-flight window animation (<= 0 means free).
    public var throttleRemaining: Double = 0
    public init() {}
}

public enum FollowDecision: Equatable, Sendable {
    case ignore
    case recordAspect(Double)
    case retryLater
    case animate(Double)
}

/// Floored modulo (lua semantics): -90 % 180 == 90.
func gorFloorMod(_ a: Int, _ b: Int) -> Int {
    let r = a % b
    return r < 0 ? r + b : r
}

/// The follow decision, in the source's guard order. On .animate the host
/// computes geometry via gorFitGeometry(demux display size, metrics) and
/// records the returned aspect as lastAspect.
public func gorFollowDecision(_ i: FollowInput) -> FollowDecision {
    if i.windowTransitionActive || i.pipWindowOwned || i.stereoWindowOwned { return .ignore }
    if i.demuxW <= 0 || i.demuxH <= 0 { return .ignore }
    var w = i.demuxW, h = i.demuxH
    if gorFloorMod(i.rotate, 180) == 90 { swap(&w, &h) }
    let a = Double(w) / Double(h)
    if let last = i.lastAspect {
        if abs(a / last - 1) < 0.02 { return .ignore }
    } else {
        return .ignore
    }
    if i.fileFresh { return .recordAspect(a) }
    if i.cropActive || i.gorcropPresent { return .ignore }
    if i.fullscreen { return .ignore }
    if !i.metricsValid { return .ignore }
    if i.throttleRemaining > 0 { return .retryLater }
    return .animate(a)
}

do {
    var k = KawLock(originalKeepsAspect: true)
    check("K1 restore before release is a no-op", k.restore() == nil && !k.released)
    check("K2 release emits setNo once", k.release() == .setNo && k.release() == nil && k.released)

    var k = KawLock(originalKeepsAspect: false)
    var cmds: [KawCommand] = []
    if let c = k.release() { cmds.append(c) }          // motion starts
    if let c = k.restore() { cmds.append(c) }          // settle
    if let c = k.restore() { cmds.append(c) }          // cancel lands too
    check("K5 settle+cancel restores once", cmds == [.setNo, .restoreOriginal] && k.originalKeepsAspect == false)
}

// F20/F21: guard ordering - a fresh file records before the crop,

check("F21 fresh file beats fullscreen and throttle", gorFollowDecision({ var i = base(); i.lastAspect = 0.5; i.fileFresh = true; i.fullscreen = true; i.throttleRemaining = 0.7; return i }()) == .recordAspect(A0))

print(failures == 0 ? "fit-follow-tests: all pass" : "fit-follow-tests: \(failures) FAIL")
exit(failures == 0 ? 0 : 1)
