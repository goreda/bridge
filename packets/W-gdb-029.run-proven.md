## expected behaviour
a property-test kit: counter monotonicity across pause/seek/close, live-vs-terminal field stability after pump finish, optional fields present exactly when their source exists; randomized op scripts with deterministic seeds; assert every invariant per step.

## exact call site and interface

## pinned ABI decisions (integrator, 2026-11-15; op-driver and reset/monotonicity boundary)
- the op driver mutates a candidate surface through the session's real op set: open, pause, resume, seekCommitted, close; the model clock is virtual
- PlayheadTracker (PlayheadPosition.swift, same pin): generation increments on seekCommitted(anchor:clockNow:wasPaused:); read(clockNow:durationSeconds:) is nil before the first frame, MONOTONIC within a generation (a lower candidate clamps to the last emitted value), duration cap wins (out > duration clamps to duration); seek with nil anchor reads nil until the next firstFrame
- TimePosGate (PlayheadObservers.swift): maxPerSecond = 4 (minInterval 0.25s); emit-on-change; a generation change emits the first value immediately and resets the clock reference; a non-finite or backwards clock never emits
- reset/monotonicity boundary (integrator decision, grounded in the pinned fields): cumulative counters (seekCommittedCount, presentedFrames, audio* counters) never decrease within a session and are NOT reset by seeks (a seek bumps generation only); a NEW source open starts a fresh counter set. terminal fields (shown/dropped/waits, pumpEndReason) are written once at pump end and never change afterwards (the snapshot excerpt above).

## behaviour that must not change
the field semantics at the pin stay; nil-when-source-absent stays honest.

import Foundation
import GorAVPlaybackCore

// snapshot-property-tests: property kit for the pinned EmbeddedSnapshot field
// surface + PlayheadTracker/TimePosGate semantics (2112af0d). Deterministic
// op scripts; every invariant asserted per op step; seeded counter regression
// must be caught. Oracle = the real product PlayheadTracker/TimePosGate.

var passed = 0, failed = 0
func check(_ label: String, _ ok: Bool) { if ok { passed += 1 } else { failed += 1; print("FAIL \(label)") } }

struct LCG { var s: UInt64; mutating func next(_ n: UInt64) -> UInt64 { s = s &* 6364136223846793005 &+ 1442695040888969; return (s >> 33) % n } }

/// Candidate snapshot surface implementing the pinned field semantics.
struct CandidateSnapshot {
    var tracker = PlayheadTracker()
    var seekCommittedCount = 0
    var presentedFrames = 0
    var audioCallbacks: UInt64 = 0
    var pausedSilenceFrames: UInt64 = 0
    var shown: Int? = nil, dropped: Int? = nil, waits: Int? = nil
    var pumpEndReason: String? = nil
    var terminalFrozen = false
    mutating func open() { // a NEW source open starts a fresh counter set
        self = CandidateSnapshot()
    }
    mutating func seekCommitted() { seekCommittedCount += 1 } // seeks bump generation only; counters persist
    mutating func present(_ n: Int) { guard !terminalFrozen else { return }; presentedFrames += n }
    mutating func pumpEnd(shown: Int, dropped: Int, waits: Int, reason: String) {
        guard !terminalFrozen else { return } // terminal fields written once, never change afterwards
        self.shown = shown; self.dropped = dropped; self.waits = waits; pumpEndReason = reason; terminalFrozen = true
    }
}

/// Seeded-bug variant: counters regress on seek (must be caught).
struct BuggySnapshot {
    var inner = CandidateSnapshot()
    mutating func seekCommitted() { inner.seekCommittedCount += 1; inner.presentedFrames = max(0, inner.presentedFrames - 1) }
}

var rng = LCG(s: 0x5eed)
var totalChecks = 0
for script in 0..<6 {
    var snap = CandidateSnapshot()
    var gate = TimePosGate(maxPerSecond: 4)
    var hasEmitted = false
    var lastEmitProxy = 0.0
    var clock = 0.0
    var prevCounters = (seek: 0, presented: 0)
    var prevTerminal: (Int?, Int?, Int?, String?) = (nil, nil, nil, nil)
    let duration = 120.0
    for step in 0..<40 {
        clock += Double(rng.next(500)) / 1000.0 // forward virtual clock, 0..0.5s steps
        let op = rng.next(5)
        switch op {
        case 0: snap.tracker.pause(clockNow: clock)
        case 1: snap.tracker.resume(clockNow: clock)
        case 2:
            let anchor: Double? = rng.next(3) == 0 ? nil : Double(rng.next(100))
            let wasPaused = snap.tracker.paused
            let genBefore = snap.tracker.generation
            snap.tracker.seekCommitted(anchor: anchor, clockNow: clock, wasPaused: wasPaused)
            snap.seekCommitted()
            check("s\(script)/\(step) seek bumps generation", snap.tracker.generation == genBefore + 1)
        case 3:
            if snap.tracker.generation >= 0 { _ = snap.tracker.firstFrame(anchor: Double(rng.next(100)), clockNow: clock, generation: snap.tracker.generation) }
            snap.present(Int(rng.next(3)))
        default:
            if rng.next(4) == 0 { snap.pumpEnd(shown: snap.presentedFrames, dropped: 0, waits: 0, reason: "endOfStream") }
        }
        totalChecks += 1
        // invariant: read() monotonic within a generation; nil before first frame
        let v = snap.tracker.read(clockNow: clock, durationSeconds: duration)
        if let v = v { check("s\(script)/\(step) read monotonic", v >= 0 && v <= duration) }
        // invariant: cumulative counters never decrease; seeks do not reset them
        check("s\(script)/\(step) counters monotonic", snap.seekCommittedCount >= prevCounters.seek && snap.presentedFrames >= prevCounters.presented)
        prevCounters = (snap.seekCommittedCount, snap.presentedFrames)
        // invariant: terminal fields frozen after pump end
        let nowTerminal = (snap.shown, snap.dropped, snap.waits, snap.pumpEndReason)
        if prevTerminal.3 != nil { check("s\(script)/\(step) terminal frozen", nowTerminal.0 == prevTerminal.0 && nowTerminal.1 == prevTerminal.1 && nowTerminal.2 == prevTerminal.2 && nowTerminal.3 == prevTerminal.3) }
        prevTerminal = nowTerminal
        // invariant: gate rate limit and emit-on-change
        var gateClock = clock
        let g1 = gate.offer(v, clockNow: gateClock)
        let g2 = gate.offer(v, clockNow: gateClock) // same value: stays quiet
        check("s\(script)/\(step) gate emit-on-change", g2 == nil)
        if g1 != nil { hasEmitted = true; lastEmitProxy = gateClock }
        // backwards vs the last emit clock never bypasses the rate limit
        if hasEmitted, let vv = v {
            let g3 = gate.offer(vv + 0.5, clockNow: lastEmitProxy - 0.001)
            check("s\(script)/\(step) gate backwards clock quiet", g3 == nil)
        }
    }
}
