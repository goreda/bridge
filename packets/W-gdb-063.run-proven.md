## excerpts (verbatim from product_sha)

```swift id=ex_fit
    public static func fit(drawableW: Int, drawableH: Int, frameW: Int, frameH: Int) -> AspectFit {
        precondition(drawableW > 0 && drawableH > 0 && frameW > 0 && frameH > 0)
        let s = min(Double(drawableW) / Double(frameW), Double(drawableH) / Double(frameH))
        let fitW = Double(frameW) * s
        let fitH = Double(frameH) * s
        let x = (Double(drawableW) - fitW) / 2
        let y = (Double(drawableH) - fitH) / 2
        return AspectFit(scaleX: fitW / Double(drawableW),
                         scaleY: fitH / Double(drawableH),
                         pixelRect: PixelRect(x: Int(x.rounded()), y: Int(y.rounded()),
                                              w: Int(fitW.rounded()), h: Int(fitH.rounded())))
    }

```

```swift id=ex_letterbox
    public static func isLetterbox(px: Int, py: Int, fit: AspectFit) -> Bool {
        px < fit.pixelRect.x || py < fit.pixelRect.y ||
        px >= fit.pixelRect.x + fit.pixelRect.w || py >= fit.pixelRect.y + fit.pixelRect.h
    }
```

```swift id=fx_aspect_fit_tests
// Sources/aspect-fit-tests/main.swift  (exit 0 = all pass)
// M2-picture/window lane (arch-196z item 3): letterbox aspect-fit harness for
// GorAVMetalPresenter/core/AspectFit.swift against fixed vectors. Pure product
// code: AspectFit.fit and AspectFit.isLetterbox run unmodified.
import GorAVMetalPresenter
#if canImport(Glibc)
import Glibc
#else
import Darwin
#endif

var failures = 0
func check(_ n: String, _ ok: Bool) { print(ok ? "PASS \(n)" : "FAIL \(n)"); if !ok { failures += 1 } }
func near(_ a: Double, _ b: Double, _ eps: Double = 1e-9) -> Bool { abs(a - b) <= eps }
func rectEq(_ r: PixelRect, _ x: Int, _ y: Int, _ w: Int, _ h: Int) -> Bool { r.x == x && r.y == y && r.w == w && r.h == h }

// V1: the v1.4 regression case - 640x360 frame in the 640x480 window must
// letterbox (AspectFit), never distort (CALayer .resize did).
let v1 = AspectFit.fit(drawableW: 640, drawableH: 480, frameW: 640, frameH: 360)
check("A1 v1.4 regression letterboxes 640x360 in 640x480", rectEq(v1.pixelRect, 0, 60, 640, 360))
check("A2 v1.4 regression NDC half extents", near(v1.scaleX, 1.0) && near(v1.scaleY, 0.75))
check("A3 letterbox band above and below, image rows inside",
      AspectFit.isLetterbox(px: 0, py: 59, fit: v1) && !AspectFit.isLetterbox(px: 320, py: 60, fit: v1) &&
      !AspectFit.isLetterbox(px: 320, py: 419, fit: v1) && AspectFit.isLetterbox(px: 320, py: 420, fit: v1))

// V2: 1920x1080 in a square 1000x1000 drawable.
let v2 = AspectFit.fit(drawableW: 1000, drawableH: 1000, frameW: 1920, frameH: 1080)
check("B1 1080p in square drawable fits width, centers height", rectEq(v2.pixelRect, 0, 219, 1000, 563))
check("B2 1080p in square drawable NDC", near(v2.scaleX, 1.0) && near(v2.scaleY, 0.5625))

// V3: portrait 360x640 in 640x480 - pillarbox.
let v3 = AspectFit.fit(drawableW: 640, drawableH: 480, frameW: 360, frameH: 640)
check("C1 portrait pillarboxes", rectEq(v3.pixelRect, 185, 0, 270, 480) && near(v3.scaleX, 0.421875) && near(v3.scaleY, 1.0))
check("C2 pillarbox bands left and right",
      AspectFit.isLetterbox(px: 184, py: 240, fit: v3) && !AspectFit.isLetterbox(px: 185, py: 240, fit: v3) &&
      !AspectFit.isLetterbox(px: 454, py: 240, fit: v3) && AspectFit.isLetterbox(px: 455, py: 240, fit: v3))

// V4: 4K in 640x480 downscales with the same geometry as V1.
let v4 = AspectFit.fit(drawableW: 640, drawableH: 480, frameW: 3840, frameH: 2160)
check("D1 4K downscale matches V1 geometry", v4.pixelRect == v1.pixelRect && near(v4.scaleX, 1.0) && near(v4.scaleY, 0.75))

// V5: exact match leaves no letterbox anywhere.
let v5 = AspectFit.fit(drawableW: 640, drawableH: 480, frameW: 640, frameH: 480)
check("E1 exact fit is full drawable", rectEq(v5.pixelRect, 0, 0, 640, 480) && near(v5.scaleX, 1.0) && near(v5.scaleY, 1.0))
check("E2 exact fit has no letterbox pixel in the drawable",
      !AspectFit.isLetterbox(px: 0, py: 0, fit: v5) && !AspectFit.isLetterbox(px: 639, py: 479, fit: v5) && !AspectFit.isLetterbox(px: 320, py: 240, fit: v5))

// V6: 1x1 frame in 1000x999 - sub-pixel center rounds away from zero.
let v6 = AspectFit.fit(drawableW: 1000, drawableH: 999, frameW: 1, frameH: 1)
check("F1 tiny frame rounds half-pixel origin away from zero", rectEq(v6.pixelRect, 1, 0, 999, 999))

// V7: product rule scales small content UP (unlike a capped contain-fit).
let v7 = AspectFit.fit(drawableW: 640, drawableH: 480, frameW: 320, frameH: 180)
check("G1 small frame scales up to fill width", rectEq(v7.pixelRect, 0, 60, 640, 360) && near(v7.scaleX, 1.0) && near(v7.scaleY, 0.75))

// V8: 4:3 content in a 16:9 drawable pillarboxes.
let v8 = AspectFit.fit(drawableW: 1920, drawableH: 1080, frameW: 1440, frameH: 1080)
check("H1 4:3 in 16:9 pillarboxes", rectEq(v8.pixelRect, 240, 0, 1440, 1080) && near(v8.scaleX, 0.75) && near(v8.scaleY, 1.0))

print(failures == 0 ? "aspect-fit-tests: all pass" : "aspect-fit-tests: \(failures) FAIL")
exit(failures == 0 ? 0 : 1)
```
