```swift id=ex_effective_video_rect
public func effectiveVideoRect(drawableW: Int, drawableH: Int, frameW: Int, frameH: Int, _ t: HostTransform) -> VideoRect? {
    guard drawableW > 0, drawableH > 0, frameW > 0, frameH > 0 else { return nil }
    let dw = Double(drawableW), dh = Double(drawableH)
    let s = min(dw / Double(frameW), dh / Double(frameH))
    let w = Double(frameW) * s * t.zoom, h = Double(frameH) * s * t.zoom
    let cx = dw / 2 + t.panX * dw
    let cy = dh / 2 - t.panY * dh
    return VideoRect(x: cx - w / 2, y: cy - h / 2, w: w, h: h)
}
```

```swift id=fx_aspect_override
// video-aspect-override (PLAN.md gap-map line 47 -> R10/G3): force a display
// aspect ratio by deriving an adjusted display width at fixed frame height,
// then reuse the pinned effectiveVideoRect for placement. Pure Swift, no
// AppKit or Metal. AUTHORED NEW 2026-11-16, M2-picture/window lane
// (arch-196z item 3).

/// Display size with the aspect override applied: width is derived from the
/// override at the frame's height, so the existing fit/transform pipeline
/// consumes it unchanged. nil on a non-finite or non-positive ratio or frame
/// size: an unrepresentable override refuses, never clamps.
public func aspectOverrideDisplaySize(frameW: Int, frameH: Int, aspect: Double) -> (w: Int, h: Int)? {
    guard aspect.isFinite, aspect > 0, frameW > 0, frameH > 0 else { return nil }
    // G0809: range-guard BEFORE the Int conversion. A finite huge aspect
    // derives an unrepresentable width; that refuses as nil, never traps.
    let dw = (Double(frameH) * aspect).rounded()
    guard dw.isFinite, dw >= 1, dw <= 9.0e18 else { return nil }
    return (Int(dw), frameH)
}

/// Keepaspect video rect under a display-aspect override; identical to
/// effectiveVideoRect when the override matches the frame's own aspect.
public func effectiveVideoRectAspectOverride(drawableW: Int, drawableH: Int,
                                             frameW: Int, frameH: Int,
                                             aspect: Double, _ t: HostTransform) -> VideoRect? {
    guard let d = aspectOverrideDisplaySize(frameW: frameW, frameH: frameH, aspect: aspect) else { return nil }
    return effectiveVideoRect(drawableW: drawableW, drawableH: drawableH,
                              frameW: d.w, frameH: d.h, t)
}
```

```swift id=fx_aspect_override_tests
func rectNear(_ r: VideoRect?, _ x: Double, _ y: Double, _ w: Double, _ h: Double) -> Bool {
    guard let r = r else { return false }
    return near(r.x, x) && near(r.y, y) && near(r.w, w) && near(r.h, h)
```
