#!/usr/bin/env python3
"""kits/bounds-table/check.py - validate cited b-gate bounds rows (stdlib only).

Every row must carry: name (non-empty str), value (positive number),
unit (s|ms), kind (hard-deadline|watchdog), citation (non-empty public
http(s) url), confidence (low|medium|high). A row missing value+unit+public
citation prints its name and fails the run. Exit 0 prints
'bounds-check: N/N rows cited, 0 uncited PASS'.
"""
import json, sys

def main():
    if len(sys.argv) != 2:
        print("usage: check.py bounds.json"); return 2
    doc = json.load(open(sys.argv[1]))
    rows = doc.get("rows")
    if not isinstance(rows, list) or not rows:
        print("bounds.json: no rows"); return 1
    bad = []
    for row in rows:
        name = row.get("name") if isinstance(row, dict) else None
        ok = (isinstance(name, str) and name.strip()
              and isinstance(row.get("value"), (int, float)) and row["value"] > 0
              and row.get("unit") in ("s", "ms")
              and row.get("kind") in ("hard-deadline", "watchdog")
              and isinstance(row.get("citation"), str) and row["citation"].startswith(("http://", "https://"))
              and row.get("confidence") in ("low", "medium", "high"))
        if not ok:
            bad.append(name if isinstance(name, str) and name else "<unnamed>")
    for name in bad:
        print(f"uncited row: {name}")
    if bad:
        print(f"bounds-check: {len(rows)-len(bad)}/{len(rows)} rows cited, {len(bad)} uncited FAIL")
        return 1
    print(f"bounds-check: {len(rows)}/{len(rows)} rows cited, 0 uncited PASS")
    return 0

if __name__ == "__main__":
    sys.exit(main())

// on the REAL embedded session - the same GorAVEmbeddedSession call surface
