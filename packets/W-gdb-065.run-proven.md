```swift id=fx_frame_time_gate
// d8 frame-time gate (PLAN.md P8): the gate verdict lives in production
// code, never asserted from test-side arithmetic. Pure Swift, no AppKit or
// Metal, so the gate is verifiable off-Mac against fixed vectors.
// Gate text (P8): "both-flags run, present-time p95 <= frame interval at
// fixture fps, 0 frames > 2x interval".
// AUTHORED NEW 2026-11-16, M2-picture/window lane (arch-196z item 3).

/// Verdict of one d8 gate evaluation.
public struct FrameTimeGateVerdict: Equatable, Sendable {
    /// Nearest-rank p95 of the per-frame present times, milliseconds.
    public let p95Ms: Double
    /// Frames whose present time exceeded twice the frame interval.
    public let over2xCount: Int
    public let frameCount: Int
    /// p95Ms <= frameIntervalMs AND over2xCount == 0.
    public let passed: Bool
    public init(p95Ms: Double, over2xCount: Int, frameCount: Int, passed: Bool) {
        self.p95Ms = p95Ms; self.over2xCount = over2xCount
        self.frameCount = frameCount; self.passed = passed
    }
}

/// d8 frame-time gate evaluator. Value type; one per fixture run.
public struct FrameTimeGate: Equatable, Sendable {
    /// Frame interval at the fixture's fps, milliseconds.
    public let frameIntervalMs: Double

    /// nil on a non-finite or non-positive fps: a fixture without a real
    /// rate cannot be gated (typed refusal, never a clamp).
    public init?(fixtureFps: Double) {
        guard fixtureFps.isFinite, fixtureFps > 0 else { return nil }
        self.frameIntervalMs = 1000.0 / fixtureFps
    }

    /// Nearest-rank p95: rank = ceil(0.95 * n) on the sorted samples.
    /// nil on an empty sample set or any non-finite sample: a gate that
    /// cannot see every frame refuses rather than rounding the hole away.
    public func evaluate(presentMs: [Double]) -> FrameTimeGateVerdict? {
        guard !presentMs.isEmpty, presentMs.allSatisfy({ $0.isFinite }) else { return nil }
        let sorted = presentMs.sorted()
        let n = sorted.count
        let rank = Int((0.95 * Double(n)).rounded(.up))
        let p95 = sorted[rank - 1]
        let limit = 2.0 * frameIntervalMs
        var over = 0
        for v in sorted where v > limit { over += 1 }
        return FrameTimeGateVerdict(p95Ms: p95, over2xCount: over, frameCount: n,
                                    passed: p95 <= frameIntervalMs && over == 0)
    }
}
```

```swift id=fx_frame_time_tests
// Sources/frame-time-tests/main.swift  (exit 0 = all pass)
// M2-picture/window lane (arch-196z item 3): d8 frame-time gate harness
// (PLAN.md P8) against GorAVPlaybackCore/FrameTimeGate.swift fixed vectors.
import GorAVPlaybackCore
#if canImport(Glibc)
import Glibc
#else
import Darwin
#endif

var failures = 0
func check(_ n: String, _ ok: Bool) { print(ok ? "PASS \(n)" : "FAIL \(n)"); if !ok { failures += 1 } }

let gate = FrameTimeGate(fixtureFps: 25)!   // interval 40 ms, 2x = 80 ms

// G1: healthy run passes.
let g1 = gate.evaluate(presentMs: [Double](repeating: 30, count: 100))!
check("G1 healthy 30 ms at 25 fps passes", g1.passed && g1.p95Ms == 30 && g1.over2xCount == 0 && g1.frameCount == 100)

// G2: the CI 42308323802 failure shape - 218 at 30 ms + 31 at 45 ms over 249.
var g2in = [Double](repeating: 30, count: 218) + [Double](repeating: 45, count: 31)
let g2 = gate.evaluate(presentMs: g2in)!
check("G2 CI failure shape fails on p95 alone", !g2.passed && g2.p95Ms == 45 && g2.over2xCount == 0)

// G3: one frame over 2x fails even when p95 is healthy.
let g3 = gate.evaluate(presentMs: [Double](repeating: 30, count: 99) + [85])!
check("G3 a single frame over 2x interval fails", !g3.passed && g3.p95Ms == 30 && g3.over2xCount == 1)

// G4: p95 exactly at the interval passes (<=).
let g4 = gate.evaluate(presentMs: [Double](repeating: 40, count: 20))!
check("G4 p95 at the interval boundary passes", g4.passed && g4.p95Ms == 40)

// G5a/G5b: nearest-rank boundary pair at n = 100.
let g5a = gate.evaluate(presentMs: [Double](repeating: 10, count: 95) + [Double](repeating: 50, count: 5))!
check("G5a rank 95 of 95x10+5x50 is 10", g5a.passed && g5a.p95Ms == 10)
let g5b = gate.evaluate(presentMs: [Double](repeating: 10, count: 94) + [Double](repeating: 50, count: 6))!
check("G5b rank 95 of 94x10+6x50 is 50 and fails", !g5b.passed && g5b.p95Ms == 50)

// G6: fixture fps must be a real positive rate.
check("G6 invalid fps refused", FrameTimeGate(fixtureFps: 0) == nil && FrameTimeGate(fixtureFps: -25) == nil && FrameTimeGate(fixtureFps: .nan) == nil && FrameTimeGate(fixtureFps: .infinity) == nil)

// G7: empty or non-finite samples refuse, never clamp.
check("G7 empty and non-finite refused", gate.evaluate(presentMs: []) == nil && gate.evaluate(presentMs: [30, .nan]) == nil && gate.evaluate(presentMs: [30, .infinity]) == nil)

// G8: 60 fps fixture - interval 16.666.. ms, 2x = 33.333.. ms.
let gate60 = FrameTimeGate(fixtureFps: 60)!
let g8 = gate60.evaluate(presentMs: [Double](repeating: 16, count: 60) + [34])!
check("G8 60 fps 2x boundary honored", !g8.passed && g8.over2xCount == 1)

print(failures == 0 ? "frame-time-tests: all pass" : "frame-time-tests: \(failures) FAIL")
exit(failures == 0 ? 0 : 1)
```
